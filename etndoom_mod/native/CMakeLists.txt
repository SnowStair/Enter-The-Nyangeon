cmake_minimum_required(VERSION 3.16)
project(ETNDOOM_M1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(ENABLE_SDL2_NET OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL2_MIXER ON CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_SampleRate ON)
set(CMAKE_DISABLE_FIND_PACKAGE_PNG ON)
set(CMAKE_DISABLE_FIND_PACKAGE_FluidSynth ON)
find_package(SDL2 2.0.14 CONFIG REQUIRED)
find_package(SDL2_mixer 2.8.1 CONFIG REQUIRED)
add_compile_options(-fwrapv -fno-strict-aliasing)
add_compile_definitions(DISABLE_SDL2NET=1)
if(M0_REFERENCE)
  add_compile_definitions(ETN_M0_REFERENCE=1)
else()
  add_compile_definitions(ETN_M0_CORE=1)
  add_compile_options(-include ${CMAKE_CURRENT_SOURCE_DIR}/m0_memory.h)
endif()
add_subdirectory(${M0_SOURCE} upstream)
target_include_directories(doom PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(M0_REFERENCE)
  target_sources(crispy-doom PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/m0_observer.c ${CMAKE_CURRENT_SOURCE_DIR}/m0_reference.c ${CMAKE_CURRENT_SOURCE_DIR}/m1_audio.c)
  target_include_directories(crispy-doom PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${M0_SOURCE}/src/doom ${M0_SOURCE}/src)
  set_target_properties(crispy-doom PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
else()
  get_target_property(core_sources crispy-doom SOURCES)
  list(FILTER core_sources EXCLUDE REGEX "i_main.c$|doom-res.rc$")
  list(TRANSFORM core_sources PREPEND "${M0_SOURCE}/src/")
  add_library(etndoom_crispy_core SHARED ${core_sources} m0_core.c m0_observer.c m0_memory.c m1_audio.c m1_save.c)
  set_source_files_properties(m0_memory.c PROPERTIES COMPILE_DEFINITIONS ETN_M0_MEMORY_IMPL=1)
  target_include_directories(etndoom_crispy_core PRIVATE ${M0_SOURCE}/src ${M0_SOURCE}/src/doom ${M0_SOURCE}/win32 ${CMAKE_BINARY_DIR}/upstream ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(etndoom_crispy_core PRIVATE doom SDL2::SDL2 SDL2_mixer::SDL2_mixer textscreen pcsound opl winmm shlwapi bcrypt)
  target_link_options(etndoom_crispy_core PRIVATE -static-libgcc)
  set_target_properties(etndoom_crispy_core PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
endif()
